165bool CntrServicoCartao::mover(const Email& usuario, const Identificador& identificador) {
166 if (this->containerCartao == nullptr || !usuarioCadastrado(usuario)) {
167 return false;
168 }
169 CartaoAtividade cartao;
170 if (!this->containerCartao->pesquisar(identificador, cartao)) {
171 return false;
172 }
173
174 std::string estadoAtual = cartao.getEstado().getValor();
175 Estado novoEstado;
176 if (estadoAtual == A_FAZER) {
177 // Entrar em FAZENDO nao pode estourar o limite de WIP do quadro.
178 Identificador idQuadro;
179 Quadro quadro;
180 if (!this->containerCartao->getQuadro(identificador, idQuadro) || !lerQuadro(usuario, idQuadro, quadro)) {
181 return false;
182 }
183 if (contarFazendo(idQuadro) >= quadro.getLimite().getValor()) {
184 return false;
185 }
186 novoEstado.setValor(FAZENDO);
187 cartao.setInicio(agora());
188 } else if (estadoAtual == FAZENDO) {
189 novoEstado.setValor(FEITO);
190 cartao.setTermino(agora());
191 } else {
192 return false; // Ja esta em FEITO.
193 }
194 cartao.setEstado(novoEstado);
195 return this->containerCartao->atualizar(cartao);
196}

222bool CntrServicoCartao::calcularMedia(const Email& usuario, const Identificador& quadro, bool ciclo, double& minutos) {
223 if (this->containerCartao == nullptr) {
224 return false;
225 }
226 Quadro lido;
227 if (!lerQuadro(usuario, quadro, lido)) {
228 return false;
229 }
230 std::vector<CartaoAtividade> cartoes;
231 this->containerCartao->listar(quadro, cartoes);
232
233 long soma = 0;
234 int concluidos = 0;
235 for (std::vector<CartaoAtividade>::size_type i = 0; i < cartoes.size(); i++) {
236 if (cartoes[i].getEstado().getValor() != FEITO) {
237 continue;
238 }
239 long comeco = (ciclo ? cartoes[i].getInicio() : cartoes[i].getEntrada()).getMinutos();
240 soma += cartoes[i].getTermino().getMinutos() - comeco;
241 concluidos++;
242 }
243 if (concluidos == 0) {
244 return false;
245 }
246 minutos = static_cast<double>(soma) / concluidos;
247 return true;
248}
