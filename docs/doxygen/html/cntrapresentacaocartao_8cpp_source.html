68void CntrApresentacaoCartao::executar(const Email& usuario) {
69 const int CRIAR = 1;
70 const int LER = 2;
71 const int ATUALIZAR = 3;
72 const int EXCLUIR = 4;
73 const int LISTAR = 5;
74 const int MOVER = 6;
75 const int TEMPO_CICLO = 7;
76 const int TEMPO_CICLO_MEDIO = 8;
77 const int LEAD_TIME = 9;
78 const int LEAD_TIME_MEDIO = 10;
79 const int WIP = 11;
80 const int VOLTAR = 12;
81
82 if (this->cntrServicoCartao == nullptr) {
83 Tela::aviso("Servico de cartoes indisponivel.");
84 return;
85 }
86 while (true) {
87 int opcao = Tela::lerOpcao("MENU DE CARTOES (usuario: " + usuario.getValor() + ")",
88 {"Criar cartao", "Ler cartao", "Atualizar cartao", "Excluir cartao",
89 "Listar cartoes do quadro", "Mover cartao", "Ler tempo de ciclo",
90 "Ler tempo de ciclo medio", "Ler lead time", "Ler lead time medio",
91 "Ler WIP", "Voltar"});
92 if (opcao == 0 || opcao == VOLTAR) {
93 return;
94 }
95 if (opcao == CRIAR) {
96 criar(usuario);
97 } else if (opcao == LER) {
98 ler(usuario);
99 } else if (opcao == ATUALIZAR) {
100 atualizar(usuario);
101 } else if (opcao == EXCLUIR) {
102 excluir(usuario);
103 } else if (opcao == LISTAR) {
104 listar(usuario);
105 } else if (opcao == MOVER) {
106 mover(usuario);
107 } else if (opcao == TEMPO_CICLO) {
108 lerTempoCiclo(usuario);
109 } else if (opcao == TEMPO_CICLO_MEDIO) {
110 lerTempoCicloMedio(usuario);
111 } else if (opcao == LEAD_TIME) {
112 lerLeadTime(usuario);
113 } else if (opcao == LEAD_TIME_MEDIO) {
114 lerLeadTimeMedio(usuario);
115 } else if (opcao == WIP) {
116 lerWIP(usuario);
117 }
118 if (Tela::entradaEncerrada()) {
119 return;
120 }
121 }
122}

126void CntrApresentacaoCartao::criar(const Email& usuario) {
127 Tela::titulo("CRIAR CARTAO");
128 Tela::info("O cartao entra na coluna A FAZER. Digite 0 em qualquer campo para cancelar.");
129 Tela::separador();
130
131 Identificador quadro;
132 if (Tela::lerCampo("Identificador do quadro", DICA_IDENTIFICADOR, quadro) != Tela::Leitura::LIDO) {
133 return;
134 }
135 Identificador identificador;
136 if (Tela::lerCampo("Identificador do cartao", DICA_IDENTIFICADOR, identificador) != Tela::Leitura::LIDO) {
137 return;
138 }
139 CartaoAtividade cartao;
140 cartao.setIdentificador(identificador);
141 if (lerDadosEditaveis(cartao) != Tela::Leitura::LIDO) {
142 return;
143 }
144 Email desenvolvedor;
145 if (Tela::lerCampo("Email do desenvolvedor", DICA_EMAIL, desenvolvedor) != Tela::Leitura::LIDO) {
146 return;
147 }
148 if (!this->cntrServicoCartao->criar(usuario, quadro, desenvolvedor, cartao)) {
149 Tela::falha("Nao foi possivel criar o cartao.",
150 {"usuario nao e GESTOR", "quadro nao existe",
151 "desenvolvedor nao existe ou nao tem papel DESENVOLVEDOR", "identificador ja existe"});
152 return;
153 }
154 Tela::sucesso("Cartao criado com sucesso na coluna A FAZER.");
155}
