61 if (this->cntrServicoProjeto == nullptr) {

65 while (true) {
66 int opcao = Tela::lerOpcao("MENU DE PROJETOS (usuario: " + usuario.getValor() + ")",
67 {"Criar projeto", "Ler projeto", "Atualizar projeto", "Excluir projeto",
68 "Listar projetos", "Voltar"});
69 if (opcao == 0 || opcao == VOLTAR) {
70 return;
71 }
72 if (opcao == CRIAR) {
73 criar(usuario);
74 } else if (opcao == LER) {
75 ler(usuario);
76 } else if (opcao == ATUALIZAR) {
77 atualizar(usuario);
78 } else if (opcao == EXCLUIR) {
79 excluir(usuario);
80 } else if (opcao == LISTAR) {
81 listar(usuario);
82 }
83 if (Tela::entradaEncerrada()) {
84 return;
