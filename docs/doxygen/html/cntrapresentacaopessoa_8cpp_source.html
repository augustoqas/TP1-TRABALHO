87void CntrApresentacaoPessoa::executar(const Email& usuario) {
88 const int CRIAR = 1;
89 const int LER = 2;
90 const int ATUALIZAR = 3;
91 const int EXCLUIR = 4;
92 const int LISTAR = 5;
93 const int VOLTAR = 6;
94
95 if (this->cntrServicoPessoa == nullptr) {
96 Tela::aviso("Servico de pessoas indisponivel.");
97 return;
98 }
99 while (true) {
100 int opcao = Tela::lerOpcao("MENU DE PESSOAS (usuario: " + usuario.getValor() + ")",
101 {"Criar pessoa", "Ler pessoa", "Atualizar pessoa",
102 "Excluir pessoa", "Listar pessoas", "Voltar"});
103 if (opcao == 0 || opcao == VOLTAR) {
104 return;
105 }
106 if (opcao == CRIAR) {
107 formularioCriacao("CRIAR PESSOA", "Pessoa criada com sucesso.", "Falha ao criar pessoa.");
108 } else if (opcao == LER) {
109 ler(usuario);
110 } else if (opcao == ATUALIZAR) {
111 atualizar(usuario);
112 } else if (opcao == EXCLUIR) {
113 excluir(usuario);
114 } else if (opcao == LISTAR) {
115 listar(usuario);
116 }
117 if (Tela::entradaEncerrada()) {
118 return;
119 }
120 }
